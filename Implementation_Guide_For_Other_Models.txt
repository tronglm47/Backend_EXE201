### 4. Controller Layer (VLivingAPI/Controllers/[Model]Controller.cs)

```csharp
[Route("api/[controller]")]
[ApiController]
public class ModelController : ControllerBase
{
    private readonly IModelService _modelService;

    public ModelController(IModelService modelService)
    {
        _modelService = modelService;
    }

    // GET: api/Model
    [HttpGet]
    public async Task<IActionResult> Get([FromQuery] ModelQueryParameters queryParams)
    {
        try
        {
            var result = await _modelService.GetAllModelAsync(queryParams);
            return Ok(new
            {
                success = true,
                data = result.Items,
                pagination = new
                {
                    currentPage = result.CurrentPage,
                    totalPages = result.TotalPages,
                    totalItems = result.TotalItems,
                    pageSize = queryParams.PageSize
                }
            });
        }
        catch (Exception)
        {
            return StatusCode(500, new { success = false, message = "Internal server error" });
        }
    }

    // GET api/Model/5
    [HttpGet("{id}")]
    public async Task<IActionResult> Get(int id, [FromQuery] string? select = null)
    {
        try
        {
            var selectedFields = new List<string>();
            if (!string.IsNullOrWhiteSpace(select))
            {
                selectedFields = select.Split(',', StringSplitOptions.RemoveEmptyEntries)
                                      .Select(x => x.Trim().ToLowerInvariant())
                                      .ToList();
            }

            var result = await _modelService.GetModelById(id, selectedFields);
            if (result == null)
            {
                return NotFound(new { success = false, message = "Model not found" });
            }

            return Ok(new { success = true, data = result });
        }
        catch (Exception)
        {
            return StatusCode(500, new { success = false, message = "Internal server error" });
        }
    }
}
```

## Lưu ý quan trọng:

1. **UnitOfWork**: Cần thêm property cho repository mới vào IUnitOfWork và UnitOfWork class
2. **Dependency Injection**: Đăng ký service trong Program.cs
3. **Naming Convention**: Giữ nhất quán tên các fields và methods
4. **Error Handling**: Sử dụng try-catch đơn giản như đã implement
5. **No Authorization**: Hiện tại không cần Authorize attribute
6. **Field Selection**: Hỗ trợ query parameter "select" để chọn fields cụ thể
